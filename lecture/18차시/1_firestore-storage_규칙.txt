[ Storage 규칙 예시 ]
match /posts/{userId}/{allPaths=**} {

      // 누구나 이미지 조회 가능
      allow read: if true;

      // 로그인한 본인만 업로드 및 수정 가능
      allow create, update: if request.auth != null
        && request.auth.uid == userId;

      // 로그인한 본인만 파일 삭제 가능
      allow delete: if request.auth != null
        && request.auth.uid == userId;
    }
